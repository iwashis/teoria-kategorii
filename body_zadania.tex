\section{Kategorie i funktory}

\begin{zadanie}
    Pokazać, że $\Rel$ jest kategorią. 
\end{zadanie}

\begin{zadanie}
Niech $A$ będzie zbiorem. Pokazać, że przyporządkowania $\Set\to \Set$ zdefiniowane na obiektach i morfizmach jak poniżej są funktorami:
\begin{itemize}
\item $X\mapsto A\times X$ oraz $(f:X\to Y)\mapsto \left ((id \times f):A\times X\to A\times Y; (a,x)\mapsto (a,f(x))\right )$,
\item $X\mapsto A+ X$ oraz $$(f:X\to Y)\mapsto (id + f):A+ X\to A+ Y; x \mapsto \left \{ \begin{array}{cc}f(x)  &\text{ jeśli }x\in X, \\ x &\text{ jeśli }x\in A \end{array}\right. $$
\item $X\mapsto X^A$ oraz 
$$
f:X\to Y\quad \mapsto \quad f^A:X^A\to Y^A; \phi\mapsto f\circ \phi. 
$$
\item $X\mapsto \mathcal{P}(X)\stackrel{def}{=} \{ A\subseteq X \}$ oraz 
$$
(f:X\to Y)\quad \mapsto \quad \mathcal{P}(f):\mathcal{P}(X)\to \mathcal{P}(Y); A\mapsto f(A). 
$$
\end{itemize}
\end{zadanie}

\end{itemize}

\begin{zadanie}\label{zad:hom-produkty}
Niech $\C$ będzie kategorią oraz niech $A\in \C$ będzie obiektem. Definiujemy przyporządkowanie $\C(A,-):\C\to \Set$ w następujący sposób:
$$
X\mapsto \C(A,X) \text{ oraz } f:X\to Y \mapsto \C(A,f):\C(A,X)\to \C(A,Y);g\mapsto f\circ g. 
$$
Pokazać, że $\C(A,-)$ jest funktorem. Udowodnić, że zachowuje on binarne produkty.\footnote{Mówimy, że funktor $F:\C\to\D$ \emph{zachowuje produkty}, jeśli $F(A\times B) \cong F(A)\times F(B)$, dla dowolnych $A,B\in \C$ dla których istnieje produkt $A\times B$ w $\C$.}
\end{zadanie}

\begin{zadanie}
    Niech $F:\C\to\C$ będzie funktorem. Pokazać, że jeśli w $\C$ istnieją binarne produkty, to w $\mathrm{Alg}(F)$ też istnieją.\footnote{Niech $F:\C\to \C$ będzie funktorem. $F$-algebrą nazywamy parę $(A,a:FA\to A)$ dla obiektu $A\in \C$. Dla dwóch $F$-algebr $(A,a:FA\to A)$, $(B,b:FB\to B)$ morfizm $h:A\to B$ nazywamy \emph{homomorfizmem} o dziedzinie $(A,a:FA\to A)$ i przeciwdziedzinie $(B,b:FB\to B)$, jeśli $h\circ a = b\circ F(h)$.  Kategorię wszystkich $F$-algebr i homomorfizmów miedzy nimi oznaczamy przez $\mathrm{Alg}(F).$} 
\end{zadanie}

(A\times B)\times C \cong A\times (B\times C).
$$
\end{zadanie}

\begin{zadanie}\label{zad:koprodukt-mon}

A^\ast + B^\ast \cong (A+B)^\ast.
$$
\end{zadanie}

\begin{zadanie}

Udowodnić, że dla dowolnej relacji równoważności $R$ na $A$ jądrem  przekształcenia ilorazowego $A\to A/R; a\mapsto [a]_R$ jest $R$.

\begin{zadanie}
Pokazać, że kategoria $\omega \mathrm{CPO}$ jest CCC, natomiast kategoria $\omega \mathrm{CPO}_\perp$ nie jest CCC.\footnote{Poset $(P,\leq)$ nazywamy $\omega CPO$, jeśli każdy przeliczalny łańcuch $x_1\leq x_2 \leq \dots$ ma supremum. Przekształcenie $f:P\to~Q$, które zachowuje porządek między dwoma posetami $(P,\leq)$ i $(Q,\leq)$, które dodatkowo są $\omega CPO$ nazywamy \emph{ciągłym}, jeśli zachowuje suprema przeliczalnych łańcuchów, tj. $f\Big(\bigvee_{i\in \mathbb{N}} x_i\Big) = \bigvee_{i} f(x_i)$ dla każdego $x_1\leq x_2\leq \ldots$. Posety, które spełniają własność $\omega CPO$, wraz z ciągłymi przekształceniami jako morfizmami, tworzą kategorię oznaczaną przez $\omega\mathrm{CPO}$. 

Poset $(P,\leq)$, który jest $\omega CPO$ nazywamy \emph{punktowym}, jeśli istnieje w nim element najmniejszy $\perp\in P$. Punktowe $\omega CPO$ tworzą kategorię, w której strzałkami są wszystkie ciągłe przekształcenia dodatkowo zachowujące element najmniejszy, tj. $h(\perp~)= \perp$. Tę kategorię oznaczamy przez $\omega \mathrm{CPO}_\perp$.}  
\end{zadanie}

\]

\[
g\cdot f \;=\; (\varepsilon_{Z\times A})^A \circ \State_A(g) \circ f.
\]
Pokazać, że tak zdefiniowane działanie jest łączne oraz $\eta_Y\cdot f = f\cdot \eta_X = f$ dla $\eta_X = \widetilde{\id_{X\times A}}$.

\begin{zadanie}\label{zad:nat-ccc}
  Niech $\mathbb{C}$ będzie kategorią kartezjańsko-domkniętą. Pokazać, że następujące rodziny:
  \begin{itemize}
    \item $\{\eta_X: X\to (X\times A)^A\}_X$, 
    \item $\{\varepsilon_X: X^A\times A\to X\}_X$,
    \item $\{\mu_X: \left ( (X\times A)^A \times A\right )^A \to (X\times A)^A \}$, gdzie 
        $\mu_X =  (\varepsilon_{X\times A})^A$. 
  \end{itemize}
  są transformacjami naturalnymi między odpowiednimi funktorami. 
\end{zadanie} 

\begin{zadanie}\label{zad:monady}

\begin{enumerate}
  \item

    \mu: \left ((\mathcal{I}d\times A)^A \times A\right)^A\implies (\mathcal{I}d\times A)^A, 
    \eta: \mathcal{I}d\implies (\mathcal{I}d\times A)^A)

  \item $(T, \mu, \eta)$, gdzie $T\in \{\mathcal{P}, M\times \mathcal{I}d\}$, oraz

\begin{zadanie}\label{zad:kleisli-sklad}
  Niech $(T,\mu,\eta)$ będzie monadą na kategorii $\mathbb{C}$. Dla $f:A\to TB, g:B\to TC$ zdefiniujmy:
  $$ g \cdot f :A\to TC; g\cdot f = \mu_C \circ Tg \circ f .$$ 
  Pokazać, że dla $f,g$ oraz $h:C\to TD$ zachodzi:
  \begin{align*}
    & (h\cdot g)\cdot f = h\cdot (g\cdot f) \text{ oraz } f\cdot \eta_A = \eta_B\cdot f = f.  
  \end{align*}
\end{zadanie}

\begin{zadanie}
 Trójkę $(T,(-)^\ast, \eta)$ nazywamy \emph{trójką Kleisliego}, jeśli
 \begin{itemize}
   \item $T:\mathbb{C}_o\to \mathbb{C}_o$ jest przyporządkowaniem,
   \item $\eta= \{\eta_X:X\to TX\}_{X\in \mathbb{C}}$ jest rodziną strzałek,
   \item $(-)^\ast$ przyporządkowuje dowolnej strzałce $f:X\to TY$ strzałkę $f^\ast :TX\to TY$, 
  \end{itemize}
 dodatkowo spełniającymi następujące równania:
 \[ 
   \eta_X^\ast = id_{TX}, \text{ oraz } f^\ast \circ \eta_X = f \text{ oraz } 
    (g^\ast \circ f)^\ast = g^\ast \circ f^\ast.
 \]
 Pokazać, że jeśli $(T, (-)^\ast, \eta)$ jest trójką Kleisliego, to $(T,\mu,\eta)$ jest monadą, gdzie $Tf = (\eta_Y\circ f)^\ast$ 
 dla każdej strzałki $f:X\to Y$ oraz
 $\mu = \{\mu_X:T^2X\to TX\}$ dla $\mu_X = (id_{TX})^\ast$. 
 Ponadto, jeśli $(T,\mu, \eta)$ jest monadą, 
 to $(T,(-)^\ast, \eta)$ jest trójką Kleisliego, gdzie $f^\ast = \mu_Y \circ Tf$ dla $f:X\to TY$. 
\end{zadanie}
